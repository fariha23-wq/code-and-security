class Solution {
    public int[] intersection(int[] nums1, int[] nums2) {
         ArrayList<Integer> ans = new ArrayList<>();
      for(int i=0;i<nums1.length;i++){
        for (int j=0;j<nums2.length;j++){
          if(nums1[i]==nums2[j]){
            ans.add(nums1[i]);
          }
        }
      
      }
      ArrayList<Integer> fans = new ArrayList<>();
      for(int k=0;k<ans.size();k++){
        if(fans.contains(ans.get(k))){
        
        }
           else{
             fans.add(ans.get(k));
           }
      }
      
      for(int m=0;m<fans.size()-1;m++){
        if(fans.get(m)<fans.get(m+1)){
        int temp=fans.get(m);
        fans.set(m, fans.get(m + 1));
      fans.set(m + 1, temp);
        }}
     int[] result = new int[fans.size()];

for (int i = 0; i < fans.size(); i++) {
    result[i] = fans.get(i);
}

return result;
    }
}
