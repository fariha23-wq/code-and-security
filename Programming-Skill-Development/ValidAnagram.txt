class Solution {
    public boolean isAnagram(String s, String t) {
        
  char[] arr = s.toCharArray();
  char[] ar = t.toCharArray();
  
    for(int i=0;i<s.length();i++){
      for(int j=0;j<t.length();j++){
        if(arr[i]==ar[j]){
        char temp=ar[j];
        ar[j]=ar[i];
        ar[i]=temp;
        }
    }
  
    }
    
 
  int count=0;
  for(int k=0;k<ar.length;k++){
    if(arr[k]==ar[k]){
    count++;
 }
  }
  if((count==ar.length)&&(count==ar.length)){
 return true;
  }
  else{
   return false;
  }
    
  }
}
