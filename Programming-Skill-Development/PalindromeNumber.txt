class Solution {
    public boolean isPalindrome(int x) {
     String st=String.valueOf(x);
 String no="";
 for(int i=0;i<st.length();i++){
no=no+st.charAt(i);
 }
 
 String re="";
 for(int i=st.length()-1;i>=0;i--){
re=re+st.charAt(i);
 }
 
 int count=0;
 for(int i=0;i<no.length();i++){
   if(no.charAt(i)==(re.charAt(i))){
   count++;
   }
 }
 if(count==no.length()){
return true;
 }
 else{
 return false;
 }   
    }
}
