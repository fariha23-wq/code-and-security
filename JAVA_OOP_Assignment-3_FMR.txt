public class Cart{
public int cartnum;
public double disap;
public double [] price=new double [3];
public String [] nm= new String [3];
public double totalprc;
public int itmnum;
public double cal;

public  void create_cart(int ro){
cartnum=ro;
}
public void cartDetails(){
System.out.println("Your cart (c"+cartnum+") :");
for (int s=0;s<itmnum;s++){
System.out.println(nm[s]+" - "+price[s]);
}
System.out.println("Discount Applied: "+disap+"%");
System.out.println("Total price: "+totalprc);
}
public void addItem(String fa, double ri){
  if (itmnum>=3){
  System.out.println("You already have 3 items on your cart");
  return;
  }
nm[itmnum]=fa;
price[itmnum]=ri;
itmnum++;
totalprc+=ri;
System.out.println(fa+" added to cart "+cartnum+".");
System.out.println("You have "+itmnum+" item(s) in your cart now.");
}
public void addItem( double meh, String za){
if (itmnum>=3){
  System.out.println("You already have 3 items on your cart");
  return;
  }
nm[itmnum]=za;
price[itmnum]=meh;
itmnum++;
totalprc+=meh;
System.out.println(za+" added to cart "+cartnum+".");
System.out.println("You have "+itmnum+" item(s) in your cart now.");
}
public void giveDiscount(double runa){
  disap=runa;
  cal=totalprc*(runa/100);
  totalprc=totalprc-cal;
  }
}
