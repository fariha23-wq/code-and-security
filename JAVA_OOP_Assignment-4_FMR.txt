public class Shape2D{
public int lngth=5;
public double ara;
public String nm;
public int othr;
public int tt;

public Shape2D (){
nm= "Square";
System.out.println("A "+nm+" has been created with length: "+lngth);
}
public void area(){
  if(nm.equals("Square")){
ara=lngth*lngth;
System.out.println("The area of the "+nm +" is: "+ara);}
  else if (nm.equals("Rectangle")){
  ara=lngth*othr;
  System.out.println("The area of the "+nm +" is: "+ara);}
  else if(nm.equals("Triangle")&& tt !=0){
  double s=(lngth+othr+tt)/2.0;
  ara=Math.sqrt(s*(s-lngth)*(s-othr)*(s-tt));
  System.out.println("The area of the "+nm +" is: "+ara);
  }
  else if (nm.equals("Triangle")){
    ara=0.5*lngth*othr;
  System.out.println("The area of the "+nm +" is: "+ara);
  }
}
public Shape2D(int aa , int bb){
lngth=aa;
othr=bb;
nm="Rectangle";
System.out.println("A "+nm +" has been created with length: "+aa +" and breadth: "+bb);
}

public class Triangle{
public int sd1;
public int sd2;
public int sd3;
public int prmtr;
public Triangle(int ro,int sh,int ni){
sd1=ro;
sd2=sh;
sd3=ni;
}
public void triangleDetails(){
System.out.println("Three sides of the triangel are: "+sd1+", "+sd2+", "+sd3);
prmtr=sd1+sd2+sd3;
System.out.println("Perimeter: "+prmtr);
}
public String printTriangleType(){
  if((sd1==sd2) &&( sd2==sd3) && (sd3==sd1)){
    return" This is an Equilateral Triangle.";
  }
  else if ((sd1!=sd2) &&( sd2!=sd3) && (sd3!=sd1)){
  return" This is a Scalene Triangle.";
  }
  else {
  return" This is a Isosceles Triangle.";
  }
}
public void compareTrinagles(Triangle kk){
  if(this==kk){
  System.out.println("These two triangle objects have the same address.");
  }
  else if(sd1==kk.sd1 && sd2==kk.sd2 && sd3==kk.sd3){
    System.out.println("Addresses are different but the sides of the triangles are equal.");
  }
  else if ((sd1+sd2+sd3)==(kk.sd1+kk.sd2+kk.sd3)){
   System.out.println("Only the perimeter of both triangles is equal.");
  }
  else {
  System.out.println("Addresses, length of the sides and perimeter all are different.");
  }
}
}
