 store= start;

 store= start;

 }

  }

public void addSong(Song xx, int pos){
if(pos==0|| start== null){
xx.next=start;
start=xx;
System.out.println(xx.snm+" added to "+name+".");
}
else{
store =start;
int cal=0;
while(store!= null && cal<pos-1){
 store=store.next;
 cal++;
}
 if(store==null){
 System.out.println("Cannot add song to Index "+pos);
 }
 else{
 xx.next=store.next;
 store.next=xx;
 System.out.println(xx.snm+" added to "+name+".");
}
  }
}

 }

 //8
public void deleteSong(String sname){
boolean dlt=false;
if(start==null){
System.out.println(sname+" not found in "+name+".");
}
if(start.snm.equals(sname)){

start=start.next;
System.out.println(sname+" deleted from "+name+ ".");
dlt=true;
}

store=start;
while(store.next!=null){
if(store.next.snm.equals(sname)){
System.out.println(sname+" deleted from "+name+".");
store.next=store.next.next;
dlt=true;
 break;
}
store=store.next;
}
if(!dlt){
System.out.println(sname+" not found in "+name+".");
}


}

public int totalSong(){
int cal=0;
store=start;
while(store!=null){
cal++;
store=store.next;
}
return cal;
}

  if(start == null){
  start=jj.start;
  }
  else{
store=start;
while(store.next!=null){
store=store.next;
}
store.next=jj.start;

}
}
