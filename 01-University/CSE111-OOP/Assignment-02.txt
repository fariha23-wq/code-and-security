public String makeOptional(){
  if (submission==true){
return "Assignment will not require submission";
  }
  else {
 return "Submission is already not required";
  }

}
}

public class Employee{
public String nm;
public int slry=30000;
public String dsgntn="junior";

public void displayInfo(){
System.out.println("Employee Name: "+nm);
System.out.println("Employee Salary: "+slry+"TK");
System.out.println("Employee Designation: "+dsgntn);

}
public void newEmployee(String nme){
nm=nme;
}
public void promoteEmployee(String prm){
dsgntn=prm;
if (prm.equals("senior")){
slry=slry+250000;
}
else if (prm.equals("lead")){
slry=slry+50000;
}
else if(prm.equals("manager")){
slry=slry+75000;
}

System.out.println(nm+"has been promoted to "+prm);
System.out.println("New Salary: "+slry+" TK");
}
public void calculateTax(){
  if(slry>50000 ) {
  double tx =(slry*0.30);
  System.out.println(nm+" Tax Amount: "+tx+"TK");
  }
else if (slry>30000){
double tx =(slry*0.10);
 System.out.println(nm+" Tax Amount: "+tx+"TK");
}
else {
 System.out.println("No need to pay tax");
}
  
}

}
