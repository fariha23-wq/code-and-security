public class Cargo{
public static double capacity=10.0;
public String nm;
public double wg;
public boolean ck=false;
public  int iD;
public static int idc;
public Cargo(String aa,double bb){
nm=aa;
wg=bb;
iD=++idc;
}
public void details(){
System.out.println("Cargo ID: "+iD+", Contents: "+nm+", Weight: "+wg+", Loaded: "+ck);
}
public static double capacity(){
return capacity;
  }
public void load(){
  if(ck=true){
if(wg<=capacity){
  capacity-=wg;
  ck=true;
System.out.println("Cargo "+iD+" loaded for transport.");
}
else{
System.out.println("Cannot load cargo, exceeds weight capacity.");
}
  }}
public void unload(){
  if(ck){
    capacity+=wg;
System.out.println("Cargo "+iD+" unloaded");
ck=false;
}
}
}
