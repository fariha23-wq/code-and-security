    sal=getBaseSalary()*(bn_s/100.0);
   f_sl=getBaseSalary()+sal;
   }
 }

tim_e=getHoursWorked();
if(tim_e>100){
setBaseSalary(getBaseSalary()+t_k);
System.out.println("$"+t_k+"Increment approved.");
//setBaseSalary(getBaseSalary()+t_k);
}
else if(tim_e>80){
setBaseSalary(getBaseSalary()+(t_k/2));
System.out.println("$"+(t_k/2)+"Increment approved.");
//setBaseSalary(getBaseSalary()+(t_k/2));
}
else{
System.out.println("Increment denied.");
}
}
public void displayInfo(){

public class CinemexTicket extends  MovieTicket {
public String t_o_m;
private static int tc;
public String stp="Regular";
public boolean ck;
public String ti;
public double m_prc;
public int ngc=15;
public int hr;
public static int getTotalTickets(){
return tc;
}
public  CinemexTicket(String mNm,String sTm,String tYp,String dTe ){
super(mNm,dTe,sTm,0);
t_o_m=tYp;
ck=false;
tc++;
  
  tID();
  for (int i=0;i<seatTypes.length;i++){
  if(stp==seatTypes[i]){
  setPrice(seatPrices[i]);
  break;
  }
}
}

public String statusm(){
  if(ck==true){
  return "Paid";
  }
  else {
  return "Not Paid";
  }
}
public String toString() {
return  "Ticket ID: "+tID()+"\n"+super.toString()+
  "\nGenre: "+t_o_m+"\nSeat Type: "+stp+"\nPrice(tk): "+getPrice()+"\nStatus: "+statusm();
    }
public String confirmPayment(){
  if(!ck){
    ck=true;
  return "Payment Successful";
  
  }
  else {
  return "Ticket price is already paid!";
  }
}
public String tID(){
ti= getMovie()+"-"+stp.charAt(0)+"-"+tc;
return ti;
}
public void calculateTicketPrice(){
  hr=0;
int k=0;
  while(k<seatTypes.length){
    if(seatTypes[k].equals(stp)){
    m_prc=seatPrices[k];
    if(hr>=18 && hr<=23){
    m_prc+=((m_prc*ngc)/100);
    }
    setPrice(m_prc);
    break;
    }
 k++; }
  System.out.println("Ticket price is calculated successfully.");
}
public  CinemexTicket(String nm,String tm,String typ,String dt,String kk ){
super(nm,dt,tm,0);
t_o_m=typ;
ck=false;
stp=kk;
tc++;
  
  tID();
  for (int i=0;i<seatTypes.length;i++){
  if(stp==seatTypes[i]){
  setPrice(seatPrices[i]);
  break;
  }
}
}
}
