public class Character {
  public String name;
  public int level;
  
  public Character(String name, int level) {
    this.name = name;
    this.level = level;
  }
  
  public void specialMove() {
    System.out.println("Character uses a generic move.");
  }
  
  public boolean equals(Object kL){
    Character mk=(Character)kL;
    if(this.name ==mk.name && this.level==mk.level){
    return true;
    }
    else{
      return false;
    }
  
  }
}

public class Bike extends Vehicle {
public String mdl;
public boolean ck=false;

public Bike(String ro,String sh,int ni, int ru,boolean na){
super(ro,ni);
super.setWheels(ru);
mdl=sh;
ck=na;
}
public void doAWheelie(){
  if(ck==true){
System.out.println("Wheelie Failed. "+getBrand()+":"+mdl+" has SideCar");
  }
  else{
System.out.println(getBrand()+":"+mdl+"is doing Wheelie!!");  
  }
}
public String toString(){
return super.toString()+"Model: "+mdl+",SideCar: "+ck;
}
}

public class Car extends Vehicle{
public String m_dl;
public int d_r;
public boolean wat_h=false;

public Car(String fa,String ri,int ha,int meh,int za,boolean bin){
super(fa,ha);
super.setWheels(za);
m_dl=ri;
d_r=meh;
wat_h=bin;
}
public void startAutoPilot(){
  if(wat_h==false){
 System.out.println(getBrand()+":"+m_dl+"has NO AutoPilot");
  }
  else {
   System.out.println(getBrand()+":"+m_dl+"AutoPilot Started");
  }
}
public String toString(){
return super.toString()+"Model: "+m_dl+", Doors: "+d_r+", AI:"+wat_h;
}
}

public class Garage {
public int cri;
public int bri;

public Car [] cars;
public Bike[] bikes;

public Garage(int cd,int mk){
cars= new Car[cd];
bikes= new Bike [mk];
System.out.println("Welcome to the Garage!");
System.out.println("Car Capacity: "+cd);
System.out.println("Bike Capacity: "+mk);
}

public void addVehicle(Vehicle jk){
  if(jk instanceof Car){
    
  if (cri<cars.length){
    cars[cri]=(Car)jk;
   cri++;
  System.out.println("A "+jk.getBrand()+" CAR has been added to the Garage");
   
  }
  else{
 System.out.println("Can't add more Cars! Capacity: "+cars.length); 
  }
  }
  else if(jk instanceof Bike){
  if (bri<bikes.length){
    bikes[bri]=(Bike)jk;
   bri++;
  System.out.println("A "+jk.getBrand()+" BIKE has been added to the Garage");
 
  }
  else{
 System.out.println("Can't add more Bikes! Capacity: "+bikes.length); 
  }
  }
  }
}
