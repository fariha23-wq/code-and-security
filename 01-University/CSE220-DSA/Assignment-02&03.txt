      return;

      }

      DNode iha  =  dh2.next;

      DNode nn =  far.next;


      DNode nm  =  iha.next;

      far.next  =  iha;

      iha.prev  =  far;

      iha.next = nn;


      if(nn!  =  null ){

      nn.prev =  iha;

      }
    
      far  =  nn;

      iha  =  nm;
      
      }

      while(ll.next!  =  null){

      ll  =  ll.next;

      }

      ll.next  =  dh1;

      dh1.prev  =  ll;

    }

public static void rangeMove(DNode dh, int start, int end) {


      if(dh  ==  null || dh.next  == dh){


      return;

      }

      DNode tt  =  dh.prev;

      DNode noWn  =  dh.next;

     
      while(noWn!  =  dh){


      DNode nxnd  =  noWn.next;


      if((int)noWn.elem  >=  start &&(int)noWn.elem  <=   end){

      noWn.prev.next  =  noWn.next;

      noWn.next.prev  =  noWn.prev;

      noWn.prev  =  tt;

      noWn.next  =  dh;

      tt.next  =  noWn;

      dh.prev  =  noWn;

      tt  =  noWn;
    
      }

      noWn =  nxnd;
     
      }
    }
