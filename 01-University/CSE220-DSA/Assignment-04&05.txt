        key  +=  "N"  ;
      
      int tot  =  0  ;

      int  ni  =  (int)  key.charAt(ros)  ;
      
      int far  =  (int)  key.charAt  (ros+1)  ;
      
      int iha  =  ni *  1000  + far  ;
      
      tot  += iha  ; 
      
      
      }
      
      return  tot % ht.length  ;
    }



public String searchHashtable( Object[] keyValuePair ){
       
      String kk  =  (String) keyValuePair[0]  ;
      
      int dX  =  hashFunction( kk )  ;
      
      PairNode nOw  =  ht[dX]  ;
      
      while( nOw  !=  null )  {
        
        if  (nOw.key.equals( kk ))  {
          
          return "Found" ;
          
        }
        
        nOw  =  nOw.next  ;
        
      }
      
      return  "Not Found"  ;

public static Stack conditionalReverse(Stack stack) {
        
      
      Stack  ros  =  new  Stack ()  ;
      
      Stack ni  =  new Stack  ()  ;
      
      while  (!  stack.isEmpty()  )  {
        
        ros.push  (stack .pop () ) ;
        
      }
      
      Integer  far  =  null  ;
      
      while (! ros .isEmpty () ){
        
        int iha  =  ros.pop()  ;
        
        if  (far  ==  null  || iha  !=  far  )  {
          
          ni.push (iha)  ;
          
        }
        
        far  =  iha  ;
        
      }
      
      return ni  ;
        
        
    }

 public static String hotPotato(String[] players, int k) {


       LinkedListQueue  qu  =  new   LinkedListQueue()  ;
      
      for  ( int ros  =  0  ; ros < players.length  ;  ros++ )  {
        
        qu.enqueue (players [ros] ) ;
        
      }
      
      while  (qu. front  != qu. rear )  {
      
        
        for( int ni  =  1  ; ni <  k  ; ni  ++)  {
          
          Object  ply  =  qu.dequeue()  ;
          
          qu.enqueue(ply ) ;
          
        }
        
        Object  rMv  =  qu.dequeue ()  ;
        
        System.out.println (rMv  + " eliminated ") ;
        
      }
      
      return (String)qu.peek();
      
      
      
    }
