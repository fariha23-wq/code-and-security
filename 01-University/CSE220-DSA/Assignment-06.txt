  {

public void insert(Task value) 
    
    {
      
      if  (  size  ==  capacity   )  
      
      {
        
        return ;  
        
      }
      
      int rosh  =  size   ;
      
      
       heap[rosh]   =   value ;
       
         
         size++  ;
       
       
       heapifyUp   (rosh)    ;
       
       
    }

    public Task extractMax() 
    
    {
        if(  size  ==  0  )  
          
        {
          
          return null  ;
          
        }
        
        int mehzain   =  0  ;
        
        Task binval  =  heap[mehzain]   ;
        
        
        int nival   =   size  -  1  ;
        
        heap  [  mehzain  ]  =  heap [  nival  ]  ; 
        
        
        size  =  nival  ;
        
        
        heapifyDown  (  mehzain  )  ;
        
        return   binval   ;
        
    
    }

    public Integer peek() 
    
    {
      
      if (size   ==    0)  
         
      {
        return  null  ;
        
        
      }
      
      else   
        
      {
        
     return  heap  [0]   ;
        
        
      }
         
    }

  
    private void heapifyUp(int index)   
    
    { 
         while  (  index  >  0 )
           
         {
           
           int   iriin   =   (index  -  1  )  /  2  ;
           
           
           if   ( heap  [  iriin  ]  .priority >=   heap  [  index  ] .priority )  
             
           {
             
             return   ;
             
             
           }
           
           
           Task   roshni   =   heap [  iriin  ]   ;
           
           
           heap  [  iriin  ]   =    heap   [ index   ]    ;
           
           
           heap  [  index  ]    =    roshni    ;
           
           
           
           index   =   iriin    ;
           
           
         }
             
       
    }
    private void heapifyDown(int index)
    
    
    {  
      while (true)      
      {
      
       int   far   =    2  *  index   +  1    ;
       
       int    iha    =   2  *  index   +  2   ;
       
       
       int   ros    =   index    ;
       
       
       if   (   far    <   size   &&  heap[  far  ] .priority  >    heap   [ ros  ] .priority )
         
       {
         
         ros   =   far   ;
         
         
       }
       
       
       if   (  iha   <  size    &&   heap  [ iha  ]  .priority  >    heap  [ ros  ]  .priority )
         
      
       {
         ros   =   iha    ;
         
         
       }
       
       
       if   (   ros   !=    index   )   
         
       {
         
        Task  ali   =   heap  [ index  ]   ;
         
        
         
         heap  [ index  ]    =   heap   [  ros  ]   ;
         
         
         
         heap   [  ros  ]    =   ali    ;
         
         
         
        index  =   ros     ;
         
         
         
       }
       
       
       else  
         
       {
         
         
         break  ;
         
       }
           
    
        
    }
