public class Assignment7{
  public static boolean isTriangle (int r,int o,int i){
    if (r+o>i && o+i>r && i+r>0){
    return true;
    }
    else {
    return false;
    }
  
  }
  public static void triArea(int r,int o, int i){
    if(isTriangle(r,o,i)==true){
      double sp=(r+o+i)/2.0;
      double ar=Math.sqrt(sp*(sp-r)*(sp-o)*(sp-i));
      System.out.println(ar);
    }
    else {
    System.out.println("Can't form triangle");
    }
  }
  public static void main(String [] args){
  boolean res = isTriangle(7,5,10);
  System.out.println( res );
  triArea(7,5,10);

  }
}

  public static boolean isPrime (int num){
    int div=1;
    int count=0;
    while(div<=num){
      if(num%div==0){
      count++;
      }
      div++;
    }
    if(count==2){
    return true;
    }
    else { 
      return false;
    }
    }
  public static boolean isPerfect (int num){
  int div=1;
  int sum=0;
  while (div<num){
    if(num%div==0){
    sum=sum+div;
    }
    div++;
  }
  if(sum==num){
  return true;
  }
  else {
  return false;
  }
  }
  public static int special_sum(int digit){
  int sum=0;
  for (int k=1;k<=digit;k++){
    if(isPrime(k)|| isPerfect(k)){
    sum=sum+k;
    }
 }
  return sum;
  }
